```python
def test_no_new_direct_reads():
    """pipeline/gen/**/*.py 禁新增注解直读（扩闸）。"""
    import glob
    files = glob.glob('pipeline/gen/**/*.py', recursive=True)
    violations = []
    for f in files:
        with open(f) as fh:
            src = fh.read()
        # 查直读（排除注释）
        code_lines = [l for l in src.split('\n') if not l.strip().startswith('#')]
        code = '\n'.join(code_lines)
        if 'json.load' in code and 'structs' in code:
            violations.append(f)
    # 白名单（待阶段 1 清理）
    whitelist = set()
    unexpected = set(violations) - whitelist
    assert not unexpected, f'新增直读: {unexpected}'
```

- [ ] **Step 2: Run test to verify it fails**

Run: `pytest tests/test_no_new_direct_read.py::test_no_new_direct_reads -v`
Expected: FAIL (violations found)
